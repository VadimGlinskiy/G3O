cmake_minimum_required(VERSION 3.20)
project(pak_replay_test LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LIBDEFLATE_BUILD_SHARED_LIB OFF CACHE BOOL "" FORCE)
set(LIBDEFLATE_BUILD_GZIP OFF CACHE BOOL "" FORCE)
add_subdirectory(../../third_party/libdeflate libdeflate)
add_executable(pak_replay_test test.cpp ../../src/pak_replay.cpp)
target_include_directories(pak_replay_test PRIVATE ../../src ../../third_party/libdeflate)
target_link_libraries(pak_replay_test PRIVATE libdeflate_static)
enable_testing()
add_test(NAME pak_replay COMMAND pak_replay_test)
